now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()


def write(path, value):
    path.write_text(json.dumps(value, indent=2) + '\n')

def profile(t, c):
    q = lambda p: json.dumps(str(p))
    reads = ['/System', '/usr', '/Library/Developer', '/Library/Apple', '/private/var/db/dyld',
             '/bin', '/sbin', '/private/preboot/Cryptexes', t, c]
    return '\n'.join(['(version 1)', '(deny default)', '(allow process*)',
        '(deny process-info* (target others))', '(allow file-read-metadata)',
        '(allow mach-lookup (global-name "com.apple.system.logger"))',
        '(deny mach-lookup (global-name "com.apple.securityd") (global-name "com.apple.security.agent"))',
        '(allow file-read* ' + ' '.join('(subpath ' + q(x) + ')' for x in reads) +
        ' (literal "/dev/null") (literal "/dev/urandom") (literal "/"))',
        '(allow file-map-executable)', '(allow file-write* (subpath ' + q(c) + ') (literal "/dev/null"))',
        '(deny file-write-unlink (literal ' + q(c) + '))', '(allow signal (target children))', ''])


def copied(source, target):
    if source.is_symlink() or not source.is_file():
        raise ValueError('input must be regular non-symlink: ' + str(source))
    data = source.read_bytes()
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data); target.chmod(0o400)
    return sha(data)


def main():
    root = Path(tempfile.mkdtemp(prefix='v4-store-binding-checks-', dir='/private/tmp'))
    t = root / 'T'; c = root / 'C'; t.mkdir(mode=0o700); c.mkdir(mode=0o700)
    print('Evidence: ' + str(root), flush=True)
    before = {}; copies = {}; dependencies = {}
    names = {str(p.relative_to(REPO)) for directory in ('framework/core', 'framework/local', 'framework/recovery', 'framework/adapters')
             for p in (REPO / directory).glob('*.py')}
    names.update(str(p.relative_to(REPO)) for p in (REPO / 'schemas').glob('*.json'))
    names.update(str(p.relative_to(REPO)) for p in (REPO / 'tests/adapters').glob('*.py'))
    names.update(('tests/lifecycle/test_lifecycle.py', 'examples/protocol/in-progress.json', 'tests/local/test_external.py', 'tests/local/test_launch_driver.py', 'adapters/PROTOCOL.md', 'adapters/codex/AGENTS.md', 'adapters/claude-code/CLAUDE.md'))
    for name in sorted(names):
        before[name] = copied(REPO / name, t / 'source' / name)
        copies['source/' + name] = before[name]
    for package in ('jsonschema', 'referencing', 'rpds', 'jsonschema_specifications', 'attrs', 'attr'):
        for source in sorted((DEPS / package).rglob('*')):
            if source.is_symlink(): raise ValueError('linked dependency')
            if source.is_file() and '__pycache__' not in source.parts:
                name = str(source.relative_to(DEPS))
                dependencies[name] = copied(source, t / 'deps' / name)
    dependencies['typing_extensions.py'] = copied(DEPS / 'typing_extensions.py', t / 'deps/typing_extensions.py')
    copies.update({'deps/' + name: value for name, value in dependencies.items()})
    git_env = dict(PATH='/usr/bin:/bin', GIT_CONFIG_NOSYSTEM='1', GIT_CONFIG_GLOBAL='/dev/null')
    head = subprocess.run(['/usr/bin/git', '-C', str(REPO), 'rev-parse', 'HEAD'],
        env=git_env, check=True, capture_output=True).stdout.decode().strip()
    (t / 'harness.py.txt').write_bytes(Path(__file__).read_bytes())
    (t / 'wrapper.py').write_text(WRAPPER); (t / 'sandbox.sb').write_text(profile(t, c))
    (t / 'marker').write_text('protected synthetic marker\n')
    (root / 'outside').write_text('synthetic outside marker\n')
    controls = {name: sha((t / name).read_bytes()) for name in ('harness.py.txt', 'wrapper.py', 'sandbox.sb', 'marker')}
    argv = ['/usr/bin/sandbox-exec', '-f', str(t / 'sandbox.sb'), '/usr/bin/python3', '-I', '-B', '-S',
            str(t / 'wrapper.py'), str(t), str(c)]
    env = dict(HOME=str(c / 'home'), TMPDIR=str(c / 'tmp'), PATH='/usr/bin:/bin:/usr/sbin:/sbin', LC_ALL='C', LANG='C')

        frozen_inputs=copies, controls=controls, evidence_level='synthetic_fixture', host_candidate_imports=0,
        provider_calls=0, scope='Store authority bridge and adapter contracts; author-run fixed processes only')
    write(t / 'launch.json', result); proc = None
    try:
        with (t / 'stdout.log').open('xb') as stdout, (t / 'stderr.log').open('xb') as stderr:
            proc = subprocess.Popen(argv, cwd=t, env=env, stdin=subprocess.DEVNULL,
                stdout=stdout, stderr=stderr, start_new_session=True)
            result['pid'] = proc.pid
            try: result.update(exit_code=proc.wait(timeout=600), outcome='observed_exit')
            except subprocess.TimeoutExpired:
                result.update(outcome='unknown', reason='test timeout; descendant stop not proven')
                os.killpg(proc.pid, signal.SIGTERM)
                try: proc.wait(timeout=5)
                except subprocess.TimeoutExpired: os.killpg(proc.pid, signal.SIGKILL); proc.wait(timeout=5)
                result['exit_code'] = proc.returncode
    except BaseException as error:

        if proc is not None and proc.poll() is None:
            os.killpg(proc.pid, signal.SIGTERM)

    return 0 if result.get('exit_code') == 0 and result['outcome'] == 'observed_exit' and result['frozen_unchanged'] else 1
