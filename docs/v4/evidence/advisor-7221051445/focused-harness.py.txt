import json,os,socket,sys,unittest
from pathlib import Path
c=json.loads(Path(sys.argv[1]).read_text())
checks={}
for name,operation in [
 ('outside_file_read',lambda:Path(c['canary']).read_text()),
 ('outside_file_write',lambda:Path(c['canary']).write_text('changed')),
 ('checkout_write',lambda:(Path(c['repo'])/'sandbox-write-canary').write_text('unexpected'))]:
 try:
  operation();checks[name]=False
 except PermissionError:
  checks[name]=True
try:
 with socket.socket() as s:
  s.connect(('127.0.0.1',9))
 checks['network_connect']=False
except PermissionError:
 checks['network_connect']=True
except OSError as e:
 checks['network_connect']=False
checks['credentials_removed']=not any(k in os.environ for k in ['GH_TOKEN','GITHUB_TOKEN','OPENAI_API_KEY','ANTHROPIC_API_KEY','SSH_AUTH_SOCK','CODEX_HOME'])
print(json.dumps({'sandbox_canaries':checks}),flush=True)
if not all(checks.values()):sys.exit(3)
names={'test_cli_failed_run_replay_stays_failed_and_does_not_launch','test_cli_success_receipt_replay_preserves_result','test_journal_only_crash_recovers_via_new_process_cli','test_journal_only_recovery_preserves_unknown_file'}
def flatten(suite):
 for item in suite:
  if isinstance(item,unittest.TestSuite):yield from flatten(item)
  else:yield item
suite=unittest.TestSuite()
for group in ['local','bootstrap']:
 found=unittest.TestLoader().discover(str(Path(c['repo'])/'tests'/group),pattern='test_*.py')
 suite.addTests(t for t in flatten(found) if t.id().split('.')[-1] in names)
r=unittest.TextTestRunner(verbosity=2).run(suite)
print(json.dumps({'run':r.testsRun,'failures':len(r.failures),'errors':len(r.errors),'skipped':len(r.skipped)}))
sys.exit(0 if r.wasSuccessful() else 1)
