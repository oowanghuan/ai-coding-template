import json,os,pathlib,socket
r={}
try:r['public_read']=pathlib.Path('public.txt').read_text()=='public fixture'
except OSError as e:r['public_read']=False
try:pathlib.Path('allowed.txt').write_text('ok');r['own_write']=True
except OSError:r['own_write']=False
for name,op in [('trusted_read',lambda:pathlib.Path(os.environ['CANARY']).read_bytes()),('trusted_write',lambda:pathlib.Path(os.environ['CANARY']).write_bytes(b'bad')),('network',lambda:socket.create_connection(('127.0.0.1',9),timeout=1))]:
 try:op();r[name]={'denied':False}
 except OSError as e:r[name]={'denied':e.errno in (1,13),'errno':e.errno}
print(json.dumps(r))
