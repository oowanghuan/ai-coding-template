class Experiment:
 def __init__(self,info):
  self.info=info;self.t=Path(info['trusted']);self.c=Path(info['candidate']);self.results=[];self.counter=0
  self.environment=env(self.c,self.c);self.environment.update(PYTHONPATH=str(REPO)+':'+str(DEPS),V4_OUTER_SEATBELT='1')
 def command(self,label,args,expected):
  self.counter+=1;argv=['/usr/bin/sandbox-exec','-f',str(self.t/'outer.sb'),PYTHON,'-B','-S']+args
  before=utc();r=invoke(argv,self.c,self.environment)
  prefix=STAGE/('%02d-'%self.counter+label);prefix.with_suffix('.stdout.log').write_bytes(r.stdout);prefix.with_suffix('.stderr.log').write_bytes(r.stderr)
  try:decision=json.loads(r.stdout)
  except Exception:decision=None
  record=dict(label=label,argv=argv,environment=self.environment,started_at=before,finished_at=utc(),exit_code=r.returncode,decision=decision,expected_status=expected,stdout_sha256=digest(r.stdout),stderr_sha256=digest(r.stderr))
  record['passed']=decision is not None and decision.get('status')==expected and r.returncode==({'passed':0,'failed':1,'unverifiable':3,'effective':0,'rejected':2}[expected])
  save(prefix.with_suffix('.json'),record);self.results.append(record)
  if not record['passed']:raise RuntimeError('unexpected CLI result: '+label+' '+str(record))
  return decision
 def quality(self,ident):
  return self.command('quality-'+ident,[str(REPO/'scripts/v4_quality.py'),'verify','--trust',str(self.t/'quality-trust.json'),'--trust-sha256',self.info['quality_trust_sha256'],'--run-id',ident], 'passed' if ident in ('success','fresh-success','mechanical') else 'failed')

  return self.command(label,[str(REPO/'scripts/v4_governance.py'),'verify','--trust',str(bundle[0]),'--trust-sha256',bundle[1],'--signoff',signoff,'--candidate',str(path),'--at',AT],expected)
