steps=[];checks=[]
def H(b):return hashlib.sha256(b).hexdigest()
def tree(p):return {str(x.relative_to(p)):H(x.read_bytes()) for x in sorted(p.rglob('*')) if x.is_file()}
def check(n,ok):assert ok,n;checks.append(n)
def run(name,args,cwd):
 p=subprocess.run(args,cwd=cwd,capture_output=True,timeout=30);stem='%02d-%s'%(len(steps)+1,name)
 for suffix,data in [('stdout',p.stdout),('stderr',p.stderr)]: (logs/(stem+'.'+suffix)).write_bytes(data)
 steps.append({'name':name,'argv':args,'cwd':str(cwd),'exit_code':p.returncode,'stdout_file':stem+'.stdout','stderr_file':stem+'.stderr','stdout_sha256':H(p.stdout),'stderr_sha256':H(p.stderr)})
 assert p.returncode==0,(name,p.returncode,p.stdout,p.stderr)
 return json.loads(p.stdout) if p.stdout.strip().startswith(b'{') else p.stdout

def cli(framework,name,root,action,*args):return run(name,[sys.executable,'-B','-S',str(framework/'scripts/v4_project.py'),action,'--root',str(root),*args],root)
def inspect(framework,name,root):return run(name,[sys.executable,'-B','-S',str(framework/'scripts/v4_local.py'),'inspect','--root',str(root)],root)
def git(name,root,*args):return run(name,['/usr/bin/git','-c','core.hooksPath=/dev/null','-c','core.fsmonitor=false',*args],root)
def plan(framework,name,root,mode):
 p=cli(framework,name,root,'plan-'+mode,'--owner','synthetic-owner','--project-id','upgrade-demo');f=BASE/(name+'.json');f.write_text(json.dumps(p,indent=2)+'\n');return p,f
def apply(framework,name,root,p,f):return cli(framework,name,root,'apply','--plan',str(f),'--plan-sha256',p['plan_sha256'])
