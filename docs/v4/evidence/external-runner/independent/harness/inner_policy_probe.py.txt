import hashlib,json,os,subprocess,time,uuid
from pathlib import Path
base=Path(__file__).resolve().parent
experiment=Path('<EXPERIMENT>');trusted=experiment/'trusted';candidate=experiment/'candidate'
scratch=trusted/'scratch/job-a';policy=scratch/'policy.sb'
paths={'trusted_config':str(trusted/'config.json'),'trusted_receipt':str(trusted/'receipts/job-a.json'),'authority':str(candidate/'project/.ai-org/state.json'),'checkout':str(Path('<REPOSITORY>/framework/local/external.py'))}

protected=[Path(v) for v in paths.values()]+[scratch/'stdout.log',scratch/'stderr.log',policy]
before={str(p):H(p.read_bytes()) for p in protected}
canary=scratch/('independent-inner-canary-'+uuid.uuid4().hex+'.txt')
probe="""import json,os,socket
from pathlib import Path
paths=json.loads(__import__('sys').argv[1]);canary=Path(__import__('sys').argv[2]);result={}
for name,path in paths.items():
 for operation in ('read','write'):
  try:
   flags=os.O_RDONLY if operation=='read' else os.O_WRONLY
   fd=os.open(path,flags|os.O_NOFOLLOW)
   try:
    if operation=='read':os.read(fd,1)
   finally:os.close(fd)
   result[name+'_'+operation]='unexpected_allowed_without_mutation'
  except PermissionError as error:result[name+'_'+operation]={'denied_errno':error.errno}
try:
 with socket.socket() as connection:connection.connect(('127.0.0.1',9))
 result['network']='unexpected_allowed'
except PermissionError as error:result['network']={'denied_errno':error.errno}

print(json.dumps(result))
"""
command=['/usr/bin/sandbox-exec','-f',str(policy),'/usr/bin/python3','-I','-B','-S','-c',probe,json.dumps(paths),str(canary)]

observed=json.loads(p.stdout);after={str(p):H(p.read_bytes()) for p in protected}
assert p.returncode==0 and not p.stderr and before==after
assert observed['own_scratch_unique_write'] and observed['credentials_absent']

(base/'inner-policy-result.json').write_text(json.dumps(record,indent=2)+'\n');print(json.dumps({'exit_code':p.returncode,'observations':observed,'protected_original_bytes_unchanged':before==after,'policy_sha256':record['policy_sha256']}))
