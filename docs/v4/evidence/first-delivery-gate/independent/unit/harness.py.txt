from pathlib import Path

REPO=Path('<workspace>')
sha=lambda raw:hashlib.sha256(raw).hexdigest()
now=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()

def write(path,value):path.write_text(json.dumps(value,indent=2)+'\n')

WRAPPER=r'''import json,os,socket,sys,unittest
from pathlib import Path
t=Path(sys.argv[1]);c=Path(sys.argv[2]);checks={}

    try:action()
    except PermissionError:checks[name]=True
    except OSError:checks[name]=False
    else:checks[name]=False
# An empty anchor is required: ENOTEMPTY would not prove unlink protection.
checks['empty_candidate_anchor']=not list(c.iterdir())
denied('candidate_anchor_unlink',lambda:os.rmdir(c))
denied('trusted_write',lambda:(t/'marker').write_text('forged'))
denied('source_write',lambda:(t/'source/scripts/run_worker_demo.py').write_text('forged'))
denied('trusted_hardlink',lambda:os.link(t/'marker',c/'alias'))

def profile(t,c):
    q=lambda p:json.dumps(str(p))
    reads=['/System','/usr','/Library/Developer','/Library/Apple','/private/var/db/dyld','/bin','/sbin','/private/preboot/Cryptexes',t,c]
    return '\n'.join(['(version 1)','(deny default)','(allow process*)','(deny process-info* (target others))',
        '(allow file-read-metadata)','(allow mach-lookup (global-name "com.apple.system.logger"))',
        '(deny mach-lookup (global-name "com.apple.securityd") (global-name "com.apple.security.agent"))',
        '(allow file-read* '+' '.join('(subpath '+q(x)+')' for x in reads)+' (literal "/dev/null") (literal "/dev/urandom") (literal "/"))',
        '(allow file-map-executable)','(allow file-write* (subpath '+q(c)+') (literal "/dev/null"))',
        '(deny file-write-unlink (literal '+q(c)+'))','(allow signal (target children))',''])


def main():
    root=Path(tempfile.mkdtemp(prefix='v4-first-delivery-tests-',dir='/private/tmp'))
    t=root/'T';c=root/'C';t.mkdir(mode=0o700);c.mkdir(mode=0o700)
    print('Evidence: '+str(t),flush=True)
    before={};copies={}
    for name in ('scripts/run_worker_demo.py','tests/worker_demo/test_first_delivery_gate.py'):
        data=(REPO/name).read_bytes();before[name]=sha(data)
        target=t/'source'/name;target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes(data);target.chmod(0o400)
        copies[str(target.relative_to(t))]=sha(data)
    (t/'harness.py.txt').write_bytes(Path(__file__).read_bytes())
    (t/'wrapper.py').write_text(WRAPPER);(t/'sandbox.sb').write_text(profile(t,c));(t/'marker').write_text('protected synthetic marker\n')
    (root/'outside').write_text('synthetic outside marker\n')
    control={name:sha((t/name).read_bytes()) for name in ('harness.py.txt','wrapper.py','sandbox.sb','marker')}
    argv=['/usr/bin/sandbox-exec','-f',str(t/'sandbox.sb'),'/usr/bin/python3','-I','-B','-S',str(t/'wrapper.py'),str(t),str(c)]
    env=dict(HOME=str(c/'home'),TMPDIR=str(c/'tmp'),PATH='/usr/bin:/bin:/usr/sbin:/sbin',LC_ALL='C',LANG='C')
    result=dict(argv=argv,environment=env,started_at=now(),source_inputs=before,frozen_inputs=copies,controls=control,evidence_level='synthetic_fixture',host_candidate_imports=0,collector_calls=0)
    write(t/'launch.json',result);proc=None
    try:
        with (t/'stdout.log').open('xb') as stdout,(t/'stderr.log').open('xb') as stderr:
            proc=subprocess.Popen(argv,cwd=t,env=env,stdin=subprocess.DEVNULL,stdout=stdout,stderr=stderr,start_new_session=True)
            result['pid']=proc.pid
            try:result.update(exit_code=proc.wait(timeout=120),outcome='observed_exit')
            except subprocess.TimeoutExpired:
                result.update(outcome='unknown',reason='test timeout; descendant stop not proven')
                os.killpg(proc.pid,signal.SIGTERM)
                try:proc.wait(timeout=5)
                except subprocess.TimeoutExpired:os.killpg(proc.pid,signal.SIGKILL);proc.wait(timeout=5)
                result['exit_code']=proc.returncode
    except BaseException as error:
        result.update(outcome='error',error_type=type(error).__name__,error=str(error))
        if proc is not None and proc.poll() is None:
            os.killpg(proc.pid,signal.SIGTERM)
            try:proc.wait(timeout=5)
            except subprocess.TimeoutExpired:os.killpg(proc.pid,signal.SIGKILL);proc.wait(timeout=5)
            result['descendant_stop']='unknown'
    finally:
        result['finished_at']=now()
        result['source_after']={name:sha((REPO/name).read_bytes()) for name in before}
        result['source_unchanged']=result['source_after']==before
        result['frozen_unchanged']=all(sha((t/name).read_bytes())==value for name,value in {**copies,**control}.items())
        result['logs']={name:sha((t/name).read_bytes()) for name in ('stdout.log','stderr.log') if (t/name).exists()}
        write(t/'result.json',result)
    print(json.dumps({'result':str(t/'result.json'),'outcome':result['outcome'],'exit_code':result.get('exit_code'),'frozen_unchanged':result['frozen_unchanged']}))
    return 0 if result.get('exit_code')==0 and result['outcome']=='observed_exit' and result['frozen_unchanged'] else 1

if __name__=='__main__':raise SystemExit(main())
