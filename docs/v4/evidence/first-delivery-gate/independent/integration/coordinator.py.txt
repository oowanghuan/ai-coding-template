def replace_existing_owned(c_root, relative, expected, replacement, allowed):
    if relative not in allowed or relative.startswith('/') or '..' in Path(relative).parts:
        raise ValueError('write outside two explicit owned fixture files')
    path=c_root/relative
    search=0x40000000 if sys.platform=='darwin' else getattr(os,'O_PATH',os.O_RDONLY)
    directory=os.open('/',search|os.O_DIRECTORY|os.O_NOFOLLOW)
    leaf=None
    try:
        for part in path.parts[1:-1]:
            child=os.open(part,search|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=directory)
            os.close(directory);directory=child
        leaf=os.open(path.name,os.O_RDWR|os.O_NOFOLLOW|os.O_NONBLOCK,dir_fd=directory)
        before=os.fstat(leaf)
        if not stat.S_ISREG(before.st_mode) or before.st_nlink!=1 or before.st_size>4*1024*1024:
            raise ValueError('bounded single-link existing fixture required')
        observed=b''
        while True:
            part=os.read(leaf,65536)
            if not part:break
            observed+=part
            if len(observed)>4*1024*1024:raise ValueError('fixture grew during mutation check')
        signature=lambda x:(x.st_dev,x.st_ino,x.st_mode,x.st_nlink,x.st_size,x.st_mtime_ns,x.st_ctime_ns)
        if observed!=expected or signature(before)!=signature(os.fstat(leaf)):
            raise ValueError('fixture changed before controlled mutation')
        if len(replacement)>4*1024*1024:raise ValueError('replacement too large')
        os.lseek(leaf,0,os.SEEK_SET)
        remaining=memoryview(replacement)
        while remaining:
            written=os.write(leaf,remaining)
            if written<=0:raise OSError('short fixture write')
            remaining=remaining[written:]
        os.ftruncate(leaf,len(replacement));os.fsync(leaf)
    finally:
        if leaf is not None:os.close(leaf)
        os.close(directory)
